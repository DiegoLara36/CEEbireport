CREATE TABLE course (
    id                            BIGSERIAL PRIMARY KEY,
    source_id                     INT NOT NULL REFERENCES source(id) ON DELETE RESTRICT,
    external_id                   TEXT NOT NULL,
    name                          TEXT NOT NULL,
    category_id                   INT REFERENCES category(id) ON DELETE SET NULL,
    description                   TEXT,
    course_url                    TEXT,
    image_url                     TEXT,
    duration_hours                NUMERIC(7,2) CHECK (duration_hours IS NULL OR duration_hours >= 0),
    duration_weeks                NUMERIC(5,2) CHECK (duration_weeks IS NULL OR duration_weeks >= 0),

    has_certificate               BOOLEAN NOT NULL DEFAULT FALSE,
    has_participation_certificate BOOLEAN NOT NULL DEFAULT FALSE,
    state                         TEXT NOT NULL DEFAULT 'ACTIVE'
                                   CHECK (state IN ('ACTIVE','INACTIVE','REMOVED')),
    canonical_hash                TEXT,
    first_seen_at                 TIMESTAMPTZ NOT NULL DEFAULT now(),
    last_seen_at                  TIMESTAMPTZ NOT NULL DEFAULT now(),
    created_at                    TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at                    TIMESTAMPTZ NOT NULL DEFAULT now(),
    CONSTRAINT uq_course_source_external UNIQUE (source_id, external_id)
);

-- 4. OFERTA (cohorte)
-- =====================================================================

CREATE TABLE course_offering (
    id                   BIGSERIAL PRIMARY KEY,
    course_id            BIGINT NOT NULL REFERENCES course(id) ON DELETE CASCADE,
    external_code        TEXT,
    modality_id          INT REFERENCES modality(id) ON DELETE SET NULL,
    start_date           DATE,
    end_date             DATE,
    enrollment_open_at   TIMESTAMPTZ,
    enrollment_close_at  TIMESTAMPTZ,
    duration_hours       NUMERIC(7,2) CHECK (duration_hours IS NULL OR duration_hours >= 0),
    duration_weeks       NUMERIC(5,2) CHECK (duration_weeks IS NULL OR duration_weeks >= 0),
    sessions_count       INT CHECK (sessions_count IS NULL OR sessions_count >= 0),
    seats_total          INT CHECK (seats_total IS NULL OR seats_total >= 0),
    seats_available      INT CHECK (seats_available IS NULL OR seats_available >= 0),
    enrollment_fee       NUMERIC(10,2) CHECK (enrollment_fee IS NULL OR enrollment_fee >= 0),
    state                TEXT NOT NULL DEFAULT 'ACTIVE'
                          CHECK (state IN ('ACTIVE','CANCELLED','FINISHED')),
    canonical_hash       TEXT,
    first_seen_at        TIMESTAMPTZ NOT NULL DEFAULT now(),
    last_seen_at         TIMESTAMPTZ NOT NULL DEFAULT now(),
    created_at           TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at           TIMESTAMPTZ NOT NULL DEFAULT now(),
    CONSTRAINT chk_offering_seats CHECK (
        seats_available IS NULL OR seats_total IS NULL OR seats_available <= seats_total
    ),
    CONSTRAINT chk_offering_dates CHECK (
        start_date IS NULL OR end_date IS NULL OR end_date >= start_date
    )
);
CREATE UNIQUE INDEX uq_offering_course_external
    ON course_offering(course_id, external_code)
    WHERE external_code IS NOT NULL;
CREATE INDEX idx_offering_course_state ON course_offering(course_id, state);
CREATE INDEX idx_offering_start        ON course_offering(start_date);
CREATE INDEX idx_offering_modality     ON course_offering(modality_id);

CREATE TABLE offering_professor (
    id           BIGSERIAL PRIMARY KEY,
    offering_id  BIGINT NOT NULL REFERENCES course_offering(id) ON DELETE CASCADE,
    professor_id BIGINT NOT NULL REFERENCES professor(id) ON DELETE RESTRICT,
    role         TEXT NOT NULL DEFAULT 'TITULAR'
                  CHECK (role IN ('TITULAR','CO_DOCENTE','INVITADO')),
    CONSTRAINT uq_offering_prof_role UNIQUE (offering_id, professor_id, role)
);
CREATE INDEX idx_offering_prof_prof ON offering_professor(professor_id);

CREATE TABLE offering_schedule (
    id           BIGSERIAL PRIMARY KEY,
    offering_id  BIGINT NOT NULL REFERENCES course_offering(id) ON DELETE CASCADE,
    day_of_week  SMALLINT NOT NULL CHECK (day_of_week BETWEEN 1 AND 7),
    start_time   TIME NOT NULL,
    end_time     TIME NOT NULL,
    CONSTRAINT chk_schedule_times CHECK (end_time > start_time)
);
CREATE INDEX idx_schedule_offering ON offering_schedule(offering_id);

-- =====================================================================
-- 5. PRECIOS
-- =====================================================================

CREATE TABLE price_snapshot (
    id               BIGSERIAL PRIMARY KEY,
    offering_id      BIGINT NOT NULL REFERENCES course_offering(id) ON DELETE CASCADE,
    currency         CHAR(3) NOT NULL DEFAULT 'PEN' CHECK (currency IN ('PEN','USD')),
    list_amount      NUMERIC(10,2) NOT NULL CHECK (list_amount >= 0),
    final_amount     NUMERIC(10,2) NOT NULL CHECK (final_amount >= 0),
    tax_included     BOOLEAN NOT NULL DEFAULT TRUE,
    valid_from       TIMESTAMPTZ NOT NULL,
    valid_to         TIMESTAMPTZ,
    is_current       BOOLEAN GENERATED ALWAYS AS (valid_to IS NULL) STORED,
    snapshot_reason  TEXT NOT NULL DEFAULT 'INITIAL'
                      CHECK (snapshot_reason IN ('INITIAL','CHANGE_DETECTED','MANUAL')),
    raw_payload      JSONB,
    created_at       TIMESTAMPTZ NOT NULL DEFAULT now(),
    CONSTRAINT chk_price_valid CHECK (valid_to IS NULL OR valid_to > valid_from)
);
CREATE UNIQUE INDEX uq_price_snapshot_current ON price_snapshot(offering_id) WHERE is_current = TRUE;
CREATE INDEX idx_price_snapshot_offering ON price_snapshot(offering_id, valid_from DESC);

CREATE TABLE offering_price_tier (
    id           BIGSERIAL PRIMARY KEY,
    offering_id  BIGINT NOT NULL REFERENCES course_offering(id) ON DELETE CASCADE,
    segment_id   INT NOT NULL REFERENCES segment(id) ON DELETE RESTRICT,
    amount       NUMERIC(10,2) NOT NULL CHECK (amount >= 0),
    currency     CHAR(3) NOT NULL DEFAULT 'PEN' CHECK (currency IN ('PEN','USD')),
    notes        TEXT,
    valid_from   TIMESTAMPTZ NOT NULL,
    valid_to     TIMESTAMPTZ,
    is_current   BOOLEAN GENERATED ALWAYS AS (valid_to IS NULL) STORED,
    created_at   TIMESTAMPTZ NOT NULL DEFAULT now(),
    CONSTRAINT chk_tier_valid CHECK (valid_to IS NULL OR valid_to > valid_from)
);
CREATE UNIQUE INDEX uq_tier_current
    ON offering_price_tier(offering_id, segment_id) WHERE is_current = TRUE;
CREATE INDEX idx_tier_offering ON offering_price_tier(offering_id);

CREATE TABLE discount (
    id           BIGSERIAL PRIMARY KEY,
    source_id    INT NOT NULL REFERENCES source(id) ON DELETE RESTRICT,
    course_id    BIGINT REFERENCES course(id) ON DELETE CASCADE,
    offering_id  BIGINT REFERENCES course_offering(id) ON DELETE CASCADE,
    segment_id   INT REFERENCES segment(id) ON DELETE SET NULL,
    kind         TEXT NOT NULL CHECK (kind IN ('PERCENT','FIXED')),
    value        NUMERIC(10,2) NOT NULL CHECK (value > 0),
    description  TEXT,
    conditions   TEXT,
    stackable    BOOLEAN NOT NULL DEFAULT FALSE,
    valid_from   DATE,
    valid_to     DATE,
    created_at   TIMESTAMPTZ NOT NULL DEFAULT now(),
    CONSTRAINT chk_discount_scope CHECK (
        (course_id IS NOT NULL AND offering_id IS NULL) OR
        (course_id IS NULL AND offering_id IS NOT NULL)
    ),
    CONSTRAINT chk_discount_pct CHECK (
        kind <> 'PERCENT' OR (value > 0 AND value <= 100)
    ),
    CONSTRAINT chk_discount_dates CHECK (
        valid_from IS NULL OR valid_to IS NULL OR valid_to >= valid_from
    )
);
CREATE INDEX idx_discount_course   ON discount(course_id);
CREATE INDEX idx_discount_offering ON discount(offering_id);
CREATE INDEX idx_discount_segment  ON discount(segment_id);

CREATE TABLE price_snapshot_discount (
    id                BIGSERIAL PRIMARY KEY,
    price_snapshot_id BIGINT NOT NULL REFERENCES price_snapshot(id) ON DELETE CASCADE,
    discount_id       BIGINT NOT NULL REFERENCES discount(id) ON DELETE CASCADE,
    applied           BOOLEAN NOT NULL DEFAULT TRUE,
    CONSTRAINT uq_ps_discount UNIQUE (price_snapshot_id, discount_id)
);

-- 10. TRIGGERS
-- =====================================================================

-- Cierre automático del "current" anterior: precio por offering
CREATE OR REPLACE FUNCTION fn_close_prev_price_snapshot()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.valid_to IS NULL THEN
        UPDATE price_snapshot
           SET valid_to = COALESCE(NEW.valid_from, now())
         WHERE offering_id = NEW.offering_id
           AND valid_to IS NULL
           AND (TG_OP = 'INSERT' OR id <> NEW.id);
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER tg_price_snapshot_close_prev
BEFORE INSERT OR UPDATE ON price_snapshot
FOR EACH ROW EXECUTE FUNCTION fn_close_prev_price_snapshot();

-- Cierre del tier anterior: (offering, segment)
CREATE OR REPLACE FUNCTION fn_close_prev_offering_price_tier()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.valid_to IS NULL THEN
        UPDATE offering_price_tier
           SET valid_to = COALESCE(NEW.valid_from, now())
         WHERE offering_id = NEW.offering_id
           AND segment_id  = NEW.segment_id
           AND valid_to IS NULL
           AND (TG_OP = 'INSERT' OR id <> NEW.id);
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER tg_offering_price_tier_close_prev
BEFORE INSERT OR UPDATE ON offering_price_tier
FOR EACH ROW EXECUTE FUNCTION fn_close_prev_offering_price_tier();

-- Cierre del plan anterior: (offering, card, installments)
CREATE OR REPLACE FUNCTION fn_close_prev_payment_plan()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.valid_to IS NULL THEN
        UPDATE payment_plan
           SET valid_to = COALESCE(NEW.valid_from, now())
         WHERE offering_id  = NEW.offering_id
           AND card_id      = NEW.card_id
           AND installments = NEW.installments
           AND valid_to IS NULL
           AND (TG_OP = 'INSERT' OR id <> NEW.id);
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER tg_payment_plan_close_prev
BEFORE INSERT OR UPDATE ON payment_plan
FOR EACH ROW EXECUTE FUNCTION fn_close_prev_payment_plan();

-- updated_at genérico
CREATE OR REPLACE FUNCTION fn_set_updated_at()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = now();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER tg_professor_updated   BEFORE UPDATE ON professor       FOR EACH ROW EXECUTE FUNCTION fn_set_updated_at();
CREATE TRIGGER tg_course_updated      BEFORE UPDATE ON course          FOR EACH ROW EXECUTE FUNCTION fn_set_updated_at();
CREATE TRIGGER tg_offering_updated    BEFORE UPDATE ON course_offering FOR EACH ROW EXECUTE FUNCTION fn_set_updated_at();
CREATE TRIGGER tg_setting_updated     BEFORE UPDATE ON system_setting  FOR EACH ROW EXECUTE FUNCTION fn_set_updated_at();

    ADD COLUMN IF NOT EXISTS consecutive_failures INT NOT NULL DEFAULT 0,

    ADD COLUMN IF NOT EXISTS last_success_at TIMESTAMPTZ,
    ADD COLUMN IF NOT EXISTS next_run_at TIMESTAMPTZ NOT NULL DEFAULT now();

    CHECK (type IN ('HTTP_ERROR', 'DOM_STRUCTURE_CHANGED', 'ZERO_RESULTS', 'VALIDATION', 'REMOVED', 'STALE', 'CONSECUTIVE_FAIL'));
